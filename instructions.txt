from dataclasses import dataclass, field
import datetime


@dataclass
class LaundryItem:
    name: str
    quantity: int
    unit_price: float
    added_at: str = field(
        default_factory=lambda: datetime.datetime.now().strftime(
            "%Y-%m-%d %H:%M:%S"
        )
    )

    @property
    def total_cost(self) -> float:
        return self.quantity * self.unit_price
